    // 构建 Schedule 对象
    final schedule = Schedule(
      // ... 其他字段 ...

          ? jsonEncode(_recurrenceRule!.toJson())
          : null,
    );

    // 发送到后端
    await scheduleService.createSchedule(schedule);
  }
}

## 🔍 判断和展示

### 判断是否为重复日程

```dart
// 方法 1: 检查 parentId（实例）
if (schedule.parentId != null) {
  print('这是重复日程的第 ${schedule.iterationIndex} 次');
}

// 方法 2: 检查 recurrence 字段（新创建）
if (schedule.recurrence != null && schedule.recurrence!.isNotEmpty) {
  print('这是一个新的重复日程');
}

// 方法 3: 使用辅助类
if (RecurrenceHelper.isRecurringInstance(schedule)) {
  print('这是重复日程的实例');
}
```

### 解析并显示重复规则

```dart
// 从 RRULE 字符串解析
final rrule = 'FREQ=WEEKLY;INTERVAL=1;BYDAY=MO,WE,FR';
final text = RecurrenceHelper.parseRRuleToText(rrule);
print(text); // 输出: 每周（周一、周三、周五）

// 从 RecurrenceRule 对象
final rule = RecurrenceRule(
  frequency: 'DAILY',
  interval: 2,
  count: 10,
);
print(rule.toDisplayText()); // 输出: 每 2 天，共 10 次
```

## 📋 完整示例

这是一个完整的日程创建页面示例：

```dart
import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:your_app/models/schedule.dart';
import 'package:your_app/models/recurrence_rule.dart';
import 'package:your_app/widgets/recurrence_editor.dart';

class CreateSchedulePage extends StatefulWidget {
  @override
  _CreateSchedulePageState createState() => _CreateSchedulePageState();
}

class _CreateSchedulePageState extends State<CreateSchedulePage> {
  final _titleController = TextEditingController();
  DateTime? _startTime;
  RecurrenceRule? _recurrenceRule;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('创建日程')),
      body: SingleChildScrollView(
        padding: EdgeInsets.all(16),
        child: Column(
          children: [
            // 标题输入
            TextField(
              controller: _titleController,
              decoration: InputDecoration(
                labelText: '标题',
                border: OutlineInputBorder(),
              ),
            ),

            SizedBox(height: 16),

            // 时间选择
            ListTile(
              title: Text('开始时间'),
              subtitle: Text(
                _startTime != null
                    ? _startTime!.toString()
                    : '未选择',
              ),
              trailing: Icon(Icons.calendar_today),
              onTap: () async {
                final date = await showDatePicker(
                  context: context,
                  initialDate: DateTime.now(),
                  firstDate: DateTime.now(),
                  lastDate: DateTime(2030),
                );
                if (date != null) {
                  setState(() => _startTime = date);
                }
              },
            ),

            SizedBox(height: 16),

            // 重复规则编辑器
            RecurrenceEditor(
              initialRule: _recurrenceRule,
              onChanged: (rule) {
                setState(() {
                  _recurrenceRule = rule;
                });
              },
            ),

            SizedBox(height: 24),

            // 保存按钮
            ElevatedButton(
              onPressed: _handleSave,
              child: Text('保存日程'),
              style: ElevatedButton.styleFrom(
                minimumSize: Size(double.infinity, 48),
              ),
            ),
          ],
        ),
      ),
    );
  }

  Future<void> _handleSave() async {
    if (_titleController.text.isEmpty || _startTime == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('请填写必填项')),
      );
      return;
    }

    final schedule = Schedule(
      id: '',
      userId: '',
      title: _titleController.text,
      startTime: _startTime!,
      allDay: false,
      status: 'pending',
      recurrence: _recurrenceRule != null
          ? jsonEncode(_recurrenceRule!.toJson())
          : null,
    );

    // TODO: 调用 API 创建日程
    // await scheduleService.createSchedule(schedule);

    Navigator.pop(context);
  }
}
```

## 🐛 常见问题

### Q1: RecurrenceRule 的 toJson() 和后端格式不一致？

A: 前端发送 JSON 格式，后端存储为 RRULE 字符串。RecurrenceRule 类已处理转换：
- 发送：使用 `toJson()` → JSON 格式
- 接收：使用 `fromRRule()` → 解析 RRULE 字符串

### Q2: 如何获取重复系列的所有实例？

A: 使用 `parentId` 分组查询：
```dart
final instances = schedules.where((s) => s.parentId == templateId).toList();
```

### Q3: 编辑重复实例时会影响其他实例吗？

A: 不会。每个实例独立存储，修改某一次不影响其他。

### Q4: 如何停止生成新实例？

A: 删除系列模板（使用 DELETE /series/:id）。
